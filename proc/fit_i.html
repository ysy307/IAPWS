        real(real64) :: v
        real(real64), intent(in) :: x, y
        real(real64) :: x1, x2, x3, x4, x5, x6, x7, x8, x9, x10, x11, x12

        if (abs(y) < 1d-10) then
            v = ieee_value(0d0, ieee_quiet_nan)
            return
        end if
        x1 = sqrt(x)
        x2 = x1 * x1
        x3 = x1 * x2
        x4 = x2 * x2
        x5 = x2 * x3
        x7 = x3 * x4
        x8 = x4 * x4
        x10 = x5 * x5
        x6 = x3 * x3
        x12 = x6 * x6
        x14 = x7 * x7
        x9 = x4 * x5
        x18 = x9 * x9

        x11 = x5 * x6
        x22 = x11 * x11
        x24 = x12 * x12
        x16 = x8 * x8
        x32 = x16 * x16
        x36 = x18 * x18
        y1 = y
        ym1 = 1d0 / y1
        y2 = y1 * y1
        ym2 = ym1 * ym1
        ym3 = ym1 * ym2
        ym4 = ym2 * ym2
        y3 = y1 * y2
        y5 = y2 * y3
        ym5 = ym2 * ym3
        ym6 = ym3 * ym3
        ym8 = ym4 * ym4
        y10 = y5 * y5
        ym10 = ym5 * ym5
        y6 = y3 * y3
        y12 = y6 * y6
        ym12 = ym6 * ym6
        v = &
            i_n(1) + i_n(2) * y1 + i_n(3) * y10 + &
            i_n(4) * x1 * ym4 + i_n(5) * x1 * ym2 + i_n(6) * x1 * ym1 + &
            i_n(7) * x1 + i_n(8) * x2 + i_n(9) * x3 * ym5 + &
            i_n(10) * x3 + i_n(11) * x4 * ym3 + i_n(12) * x4 * ym2 + &
            i_n(13) * x4 * ym1 + i_n(14) * x5 * ym6 + i_n(15) * x5 * ym1 + &
